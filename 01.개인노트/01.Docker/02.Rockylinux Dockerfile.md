- [start.sh](http://start.sh)
    
    ```
    #!/bin/bash
    # 1. SSH 데몬을 실행합니다.
    # -D: 데몬 모드 비활성화 (하지만 컨테이너에서는 background로 실행하도록 조정)
    # &를 사용하여 백그라운드로 실행
    /usr/sbin/sshd -D & 
    
    # 2. NGINX를 포그라운드로 실행합니다.
    # 이 프로세스가 컨테이너의 PID 1 역할을 대신 수행하며,
    # 이 프로세스가 살아있는 동안 컨테이너가 유지됩니다.
    exec nginx -g 'daemon off;' & 
    
    exec java -jar /usr/share/java/jenkins.war
    ```
    
- dockerfile
    
    ```docker
    # 베이스 이미지로 Rocky Linux 9 사용
    FROM rockylinux/rockylinux:9
    # 설명: 컨테이너 이미지의 기반(베이스)으로 'rockylinux/rockylinux:10' 이미지를 사용합니다.
    # Rocky Linux는 RHEL(Red Hat Enterprise Linux)의 무료 복각(rebuild) 버전입니다.
    
    # 패키지 업데이트 및 OpenSSH 서버 설치
    RUN dnf update -y && \
    # 설명: DNF(Dandified Yum) 패키지 관리자를 사용하여 시스템의 모든 설치된 패키지를 최신 버전으로 업데이트합니다.
        dnf install -y openssh-server && \
    # 설명: SSH(Secure Shell)를 통해 컨테이너에 원격으로 접속할 수 있도록 OpenSSH 서버를 설치합니다.
        echo 'root:win1234' | chpasswd && \ 
    # 설명: 'root' 사용자에게 비밀번호 'win1234'를 설정합니다. (보안상 약한 비밀번호는 실제 환경에서 사용하지 않아야 합니다.)
        sed -i 's/#PermitRootLogin prohibit-password/PermitRootLogin yes/' /etc/ssh/sshd_config && \
    # 설명: `/etc/ssh/sshd_config` 파일에서 루트(root) 사용자의 SSH 접속을 허용하도록 설정합니다.
        sed -i 's/PasswordAuthentication no/PasswordAuthentication yes/' /etc/ssh/sshd_config && \
    # 설명: SSH 접속 시 비밀번호 인증을 허용하도록 설정합니다. (기본적으로는 키 기반 인증을 선호할 수 있습니다.)
        mkdir /run/sshd
    # 설명: OpenSSH 서버 실행에 필요한 `/run/sshd` 디렉토리를 생성합니다.
    
    # 호스트 키를 생성하고 필요한 권한을 설정합니다.
    RUN ssh-keygen -A
    # 설명: SSH 서버가 클라이언트에게 자신을 인증하기 위해 필요한 **호스트 키(Host Keys)**를 자동으로 생성합니다.
    
    # 기본 패키지 설치
    RUN dnf install net-tools -y 
    # 설명: 네트워크 진단 도구인 'net-tools' (예: netstat, ifconfig 등)를 설치합니다.
    RUN dnf install ncurses -y
    # 설명: 터미널 기반 애플리케이션(예: top, htop, 일부 대화형 설치 프로그램)에 필요한 'ncurses' 라이브러리를 설치합니다.
    RUN dnf install wget -y
    # 설명: 웹에서 파일을 다운로드하는 'wget' 유틸리티를 설치합니다.
    RUN dnf install vim -y
    # 설명: 텍스트 편집기 'Vim'을 설치합니다.
    RUN dnf install git -y
    # # 설명: 소스 코드 버전 관리 시스템인 'Git'을 설치합니다.
    # RUN dnf install nginx -y
    # 설명: 웹 서버/리버스 프록시 서버인 'Nginx'를 설치합니다.
    RUN curl -o- <https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.7/install.sh> | bash && source ~/.bashrc
    # 설명: Node Version Manager (**NVM**) 설치 스크립트를 다운로드하여 실행합니다. 이를 통해 컨테이너 내에서 여러 버전의 Node.js를 관리하고 설치할 수 있습니다. 설치 후 환경 변수를 적용합니다.
    
    # node 설치
    RUN dnf install -y nodejs
    
    # jdk 설치 (재시도)
    RUN dnf install -y java-17-openjdk
    
    #jdk 설치
    #RUN dnf install -y java-21-openjdk-headless.x86_64
    # 설명: Java 21의 OpenJDK 버전 중, 그래픽 환경(GUI) 없이 서버 환경에 적합한 **헤드리스** 버전을 설치합니다.
    
    # # 젠킨스 설치
    RUN dnf install -y 'dnf-command(config-manager)'
    # 설명: DNF의 `config-manager` 명령(리포지토리 관리 기능)을 설치합니다.
    RUN rpm --import <https://pkg.jenkins.io/redhat-stable/jenkins.io-2023.key>
    # 설명: Jenkins의 공식 GPG 키를 시스템에 추가하여 Jenkins 패키지의 신뢰성을 확인합니다.
    RUN dnf config-manager --add-repo <https://pkg.jenkins.io/redhat-stable/jenkins.repo>
    # 설명: Jenkins의 안정화 버전(stable) 리포지토리를 DNF 설정에 추가합니다.
    RUN dnf install fontconfig -y
    # 설명: **Jenkins 시작 오류 방지**를 위해 폰트 구성 유틸리티인 **fontconfig**를 설치합니다. (이전 로그 문제 해결을 위한 필수 단계입니다.)
    RUN dnf install -y jenkins
    # 설명: Jenkins 자동화 서버를 설치합니다.
    
    # start.sh 파일을 이미지에 복사합니다.
    COPY start.sh /usr/local/bin/start.sh
    # 설명: 빌드 컨텍스트(Dockerfile이 있는 디렉토리)에 있는 `start.sh` 파일을 컨테이너의 `/usr/local/bin/` 경로로 복사합니다. 이 스크립트는 컨테이너가 시작될 때 실행될 것입니다.
    
    # 실행 권한을 부여합니다.
    RUN chmod +x /usr/local/bin/start.sh
    # 설명: 복사된 `start.sh` 파일에 실행 권한(Executable permission)을 부여합니다.
    
    # 포트 노출
    EXPOSE 22
    # 설명: SSH 접속을 위한 표준 포트(22번)를 외부에 노출합니다.
    EXPOSE 80
    # 설명: Nginx 웹 서버의 기본 HTTP 포트(80번)를 외부에 노출합니다.
    EXPOSE 3000
    # 설명: 일반적인 Node.js/웹 애플리케이션에서 자주 사용하는 포트(3000번)를 노출합니다.
    EXPOSE 5000
    # 설명: 또 다른 애플리케이션 또는 서비스에서 사용할 수 있는 포트(50000번)를 노출합니다.
    EXPOSE 8080
    # 설명: **Jenkins의 기본 웹 UI 포트** 및 Tomcat 등 다른 웹 서버의 표준 포트(8080번)를 외부에 노출합니다.
    
    RUN (cd /lib/systemd/system/sysinit.target.wants/; for i in *; do [ $i == \
    systemd-tmpfiles-setup.service ] || rm -f $i; done); \
    rm -f /lib/systemd/system/multi-user.target.wants/*;\
    rm -f /etc/systemd/system/*.wants/*;\
    rm -f /lib/systemd/system/local-fs.target.wants/*; \
    rm -f /lib/systemd/system/sockets.target.wants/*udev*; \
    rm -f /lib/systemd/system/sockets.target.wants/*initctl*; \
    rm -f /lib/systemd/system/basic.target.wants/*;\
    rm -f /lib/systemd/system/anaconda.target.wants/*;
    
    # 컨테이너 시작 시 start.sh 스크립트 실행 
    # CMD ["/usr/local/bin/start.sh","/usr/sbin/init"]
    CMD ["/usr/local/bin/start.sh"]
    # 설명: 컨테이너가 시작(Run)될 때 실행될 **기본 명령**을 정의합니다. `/usr/local/bin/start.sh` 스크립트를 실행합니다.
    ```
